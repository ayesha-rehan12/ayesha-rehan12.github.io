    // ✅ Helper to read ?id= parameter from URL
    function getQueryParam(name) {
      const urlParams = new URLSearchParams(window.location.search);
      return urlParams.get(name);
    }

    // ✅ Main logic
    $(document).ready(() => {
      const summaryId = getQueryParam("id");
      console.log("Summary ID:", summaryId);
      if (summaryId) loadComponents(summaryId);
    });

    // ✅ Load all components for given summary
    function loadComponents(summaryId) {
      webapi.safeAjax({
        type: "GET",
        url: `/_api/dev_components?$select=dev_componentid,dev_basevalue,dev_name,dev_consumedunit,dev_unit,dev_utilization&$filter=_dev_specification_value eq ${summaryId}`,
        contentType: "application/json",
        headers: { "Prefer": "odata.include-annotations=*" },
        success: function (data) {
          const components = data.value;
          if (!components.length) return;

          // Create table header row (horizontal)
          const headerRow = $("<tr><th>Utilization</th></tr>");
          components.forEach(c => {
            headerRow.append(`<th>${c.dev_utilization || "0"}</th>`);
          });
          $("#componentHeader").html(headerRow);

          // Add static property rows (Utilization, Unit, Consumed, Base)
          addHorizontalRow("Unit", components.map(c => c.dev_unit?? " "));
          addHorizontalRow("Limit", components.map(c => c.dev_consumedunit ?? " "));
          addHorizontalRow("Component", components.map(c => c.dev_name ?? " "));
          addHorizontalRow("Base Value", components.map(c => c.dev_basevalue ?? " "));

          // Then load related calculations for each component
          loadAllCalculations(components);
        },
        error: function (xhr) {
          console.error("Error loading components:", xhr);
        }
      });
    }

    // ✅ Helper to append a horizontal row
    function addHorizontalRow(label, values) {
      const tr = $("<tr></tr>");
      tr.append(`<td><strong>${label}</strong></td>`);
      values.forEach(v => tr.append(`<td>${v}</td>`));
      $("#componentBody").append(tr);
    }

    // ✅ Load related calculations for each component and align horizontally
// ✅ Load related calculations for each component
function loadAllCalculations(components) {
  let completed = 0;
  const allCalcMap = {}; // componentid → calculations array

  components.forEach(c => {
    webapi.safeAjax({
      type: "GET",
      url: `/_api/dev_calculations?$select=dev_finalvalue,dev_year&$filter=_dev_component_value eq ${c.dev_componentid}`,
      contentType: "application/json",
      headers: { "Prefer": "odata.include-annotations=*" },
      success: function (data) {
        allCalcMap[c.dev_componentid] = data.value || [];
        completed++;
        if (completed === components.length) {
          renderCalculations(components, allCalcMap);
        }
      },
      error: function () {
        completed++;
        if (completed === components.length) {
          renderCalculations(components, allCalcMap);
        }
      }
    });
  });
}

// ✅ Render dev_calculation horizontally below Utilization section
function renderCalculations(components, allCalcMap) {
  // Section header row
  $("#componentBody").append(`<tr><th colspan="${components.length + 1}">Utilisation Data & Calculations</th></tr>`);

  // Determine the maximum number of calculation rows across components
  const maxCalcs = Math.max(...Object.values(allCalcMap).map(arr => arr.length || 0));

  // Render each calculation row
  for (let i = 0; i < maxCalcs; i++) {
    const tr = $("<tr></tr>");
    let firstYear = " "; // default year

    // Helper function to format date as MM-DD-YYYY
    const formatDate = (dateStr) => {
      if (!dateStr) return " ";
      const date = new Date(dateStr);
      if (isNaN(date)) return dateStr; // return original if not valid date
      const mm = String(date.getMonth() + 1).padStart(2, "0");
      const dd = String(date.getDate()).padStart(2, "0");
      const yyyy = date.getFullYear();
      return `${mm}-${dd}-${yyyy}`;
    };

    components.forEach((c, idx) => {
      const calc = allCalcMap[c.dev_componentid]?.[i];
      if (idx === 0 && calc && calc.dev_year) {
        firstYear = formatDate(calc.dev_year);
      }
    });

    // First column: formatted dev_year
    tr.append(`<td><strong>${firstYear}</strong></td>`);

    // Next columns: dev_finalvalue for each component
    components.forEach(c => {
      const calc = allCalcMap[c.dev_componentid]?.[i];
      tr.append(`<td>${calc ? (calc.dev_finalvalue ?? " ") : " "}</td>`);
    });

    $("#componentBody").append(tr);
  }
}


    // ✅ Safe Web API helper for Power Pages
    (function (webapi, $) {
      function safeAjax(ajaxOptions) {
        var deferredAjax = $.Deferred();
        shell.getTokenDeferred().done(function (token) {
          ajaxOptions.headers = ajaxOptions.headers || {};
          ajaxOptions.headers["__RequestVerificationToken"] = token;
          $.ajax(ajaxOptions)
            .done((data, textStatus, jqXHR) => deferredAjax.resolve(data, textStatus, jqXHR))
            .fail(deferredAjax.reject);
        }).fail(() => deferredAjax.rejectWith(this, arguments));
        return deferredAjax.promise();
      }
      webapi.safeAjax = safeAjax;
    })(window.webapi = window.webapi || {}, jQuery);
